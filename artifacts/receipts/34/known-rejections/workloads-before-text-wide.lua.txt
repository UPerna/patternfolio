-- Reproducible inputs, not an alternative geometry implementation. Every case
-- uses the public definition/specification/document/planner/recipe pipeline.
local P,Q,C,D,L=require('pattern'),require('pattern_recipe'),require('pattern_cli_core'),require('pattern_document'),require('pattern_planner')
local M={VERSION='performance-workloads-v1'}
local levels={'small','medium','large'}
local families={
 {'floral-growth','botanical-grown'}, {'knot-topology','celtic'},
 {'circle-packing','freeform'}, {'substitution','substitution'},
 {'projection','penrose'}, {'optimization','escher'},
 {'mosaic-packing','mosaic'}, {'text-shaping','micrography'},
 {'example-synthesis','decobrush'}}
local function read(p)local f=assert(io.open(p,'rb'));local b=f:read('*a');f:close();return b end
function M.catalog()
 local a=P.array();for _,v in ipairs(families)do for i,level in ipairs(levels)do a[#a+1]={id=v[1]..'-'..level,class=v[1],family=v[2],size=level,size_index=i}end end;return a
end
local function scaffold(n)
 local v,e=P.array(),P.array();local function id(x,y)return 'v'..x..'_'..y end
 for y=1,n do for x=1,n do v[#v+1]={id=id(x,y),point={2*x-1,2*y-1}}
  if x>1 then e[#e+1]={id='h'..x..'_'..y,from=id(x-1,y),to=id(x,y)}end
  if y>1 then e[#e+1]={id='v'..x..'_'..y,from=id(x,y-1),to=id(x,y)}end
 end end
 return {vertices=v,edges=e,open_sectors=P.array()}
end
local function disk(radius)
 local faces=P.array();local function inside(x,y)return math.max(math.abs(x),math.abs(y),math.abs(x+y))<=radius end
 local function id(x,y)return 'p'..x..'_'..y end
 local function triangle(a,b,c)
  if inside(a[1],a[2])and inside(b[1],b[2])and inside(c[1],c[2])then faces[#faces+1]={id(a[1],a[2]),id(b[1],b[2]),id(c[1],c[2])}end
 end
 for y=-radius,radius do for x=-radius,radius do
  triangle({x,y},{x+1,y},{x,y+1});triangle({x+1,y},{x+1,y+1},{x,y+1})
 end end
 return {faces=faces}
end
function M.recipe(id)
 local row;for _,x in ipairs(M.catalog())do if x.id==id then row=x;break end end;assert(row,'unknown performance workload:'..tostring(id))
 local base=Q.decode(read('examples/cli/'..row.family..'.recipe'));local _,definition=C.registry(base.definition.id)
 local i=row.size_index;local params=P.plain(base.design.parameters);local inputs=P.plain(base.design.inputs);local width,height
 if row.class=='floral-growth'then local scale=({.25,.5,1})[i];width,height=21*scale,29.7*scale
 elseif row.class=='knot-topology'then local n=({3,5,8})[i];inputs={scaffold=scaffold(n)};width,height=2*n,2*n
 elseif row.class=='circle-packing'then inputs={complex=disk(({1,2,4})[i]),panel={.3,.3,5.7,5.7}};params={tolerance=1e-11,max_iterations=600};width,height=6,6
 elseif row.class=='substitution'then params.root_depth=({2,3,4})[i];params.leaf_step=.25;params.whole_root=true;width=2^(params.root_depth+1)*params.leaf_step;height=width
 elseif row.class=='projection'then width=({2,4,8})[i];height=({2,3,6})[i]
 elseif row.class=='optimization'then params.steps=({24,120,360})[i];width,height=4.2,3.2
 elseif row.class=='mosaic-packing'then params.count=({24,96,192})[i];params.iterations=({4,12,24})[i];params.sample_step=({.24,.16,.12})[i];width,height=6,4
 elseif row.class=='text-shaping'then width=13;height=({3.5,7,18})[i];inputs.text=string.rep('A fine line follows the field. ',({4,14,45})[i]):sub(1,-2);inputs.panel={{.5,.52},{12.4,1.2},{12.3,height-.6},{.8,height-1.28}}
 elseif row.class=='example-synthesis'then params.subdivisions=({4,12,24})[i];params.rows=({9,17,33})[i];width,height=4,1.5 end
 local id='performance/'..row.id
 local extensions
 if row.class=='floral-growth'then
  extensions={legacy_geometry=P.plain(base.design.extensions.legacy_geometry)}
  local scale=({.25,.5,1})[i]
  for _,key in ipairs{'pw','ph','textleft','textright','ytop','ybot','colright','headrule','footy','spilly','actual_left','actual_right'}do
   if extensions.legacy_geometry[key]~=nil then extensions.legacy_geometry[key]=extensions.legacy_geometry[key]*scale end
  end
  extensions.legacy_geometry.layout.density=.3
  if base.design.extensions.legacy_capture_style then extensions.legacy_capture_style=P.plain(base.design.extensions.legacy_capture_style)end
 end
 local specification=P.specification({id=id,definition_ref={id=definition.id,version=definition.version},seed=base.design.seed,extensions=extensions,
  parameters=params,inputs=inputs,style=P.plain(base.design.style),constraints=P.plain(base.design.constraints),
  budget={objects=100000,operations=48000000,bytes=64*1024*1024},document={unit='cm',bounds={0,0,width,height},protected_regions=P.array()}},definition)
 local doc=D.new({id=id..'/document',unit='cm',frame={id='performance/world',origin={0,0}},twoside=false,
  pages={{id='page',number=1,size={width,height},origin={0,0},trim={0,0,width,height},protected_regions=P.array()}},spreads=P.array()})
 local original=base.view.extensions and base.view.extensions.document_planning
 local capability=original and L.import_profile(definition,P.plain(original.profile))or L.profile(definition,{mode='adaptive',generation_region='media',construction_scale=64})
 local options={crop={0,0,width,height}}
 if capability.mode=='fixed_frame'then options.tile_policy='clipped_tiles'end
 local recipe,problem=L.recipe(definition,specification,doc,'page',capability,{},options)
 assert(recipe,problem and P.serialize(problem));return recipe,row
end
return M
